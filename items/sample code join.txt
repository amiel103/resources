
DROP TABLE IF EXISTS Customers CASCADE;
DROP TABLE IF EXISTS Sales CASCADE;
CREATE TABLE Customers (
  	customer_id INTEGER PRIMARY KEY,
  	name TEXT NOT NULL,
  	address TEXT NOT NULL,
	gender TEXT NOT NULL,
	salary Integer
);

SELECT 
	C.address , count(*)
FROM Customers C
join Sales S on C.customer_id = S.customer_id
group by C.address;


SELECT 
	C.gender , sum(S.total)
FROM Customers C
join Sales S on C.customer_id = S.customer_id
group by C.gender;

SELECT 
	S.item , avg(C.salary)
FROM Customers C
join Sales S on C.customer_id = S.customer_id
where S.item = 'Iphone'
group by S.item;
